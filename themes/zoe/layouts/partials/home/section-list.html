{{- /* home/section-list.html — Reusable home section listing Parameters: .section — Hugo section name (e.g. "blog", "projects") .title — Display title (falls back to section _index.md title) .url — Link for "View all" (falls back to /
/) .count — Number of items to show (default: 5) .showDate — bool .showReading — bool .showCategories — bool Info (title, description) is pulled from the section's _index.md when available. */ -}} {{- $section := .section }} {{- $count := .count | default 5 }} {{- $showDate := .showDate | default false }} {{- $showReading := .showReading | default false }} {{- $showCategories := .showCategories | default false }} {{- $mode := .mode | default "pages" }} {{- $sectionPage := site.GetPage (printf "/%s" $section) }} {{- if not $sectionPage }}{{- return }}{{- end }} {{- /* Pull title and URL from _index.md if not overridden */}} {{- $title := .title | default $sectionPage.Title }} {{- $url := .url | default (printf "/%s/" $section) }} {{- /* Check if section is enabled in config */}} {{- $enabledKey := printf "enable%s" ($section | title) }} {{- $enabled := index site.Params.sections $enabledKey }} {{- if eq $enabled false }}{{- return }}{{- end }} {{- /* Collect items — pages mode for blog/projects, sections mode for notes */}} {{- $items := slice }} {{- if eq $mode "sections" }} {{- $items = $sectionPage.Sections.ByWeight }} {{- $items = first $count $items }} {{- else }} {{- $pages := where site.RegularPages "Section" $section }} {{- $pages = sort $pages "Date" "desc" }} {{- $pages = first $count $pages }} {{- $items = $pages }} {{- end }} {{- if not $items }}{{- return }}{{- end }}
